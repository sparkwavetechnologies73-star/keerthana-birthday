/* FIREWORKS */
const canvas=document.getElementById('fireworks');
const ctx=canvas.getContext('2d');
let particles=[];
function resizeCanvas(){
canvas.width=window.innerWidth*devicePixelRatio;
canvas.height=window.innerHeight*devicePixelRatio;

function burst(x,y){
const colors=['#ff6f9e','#ffd166','#ffffff','#ed8db4','#c9a2ff'];
for(let i=0;i<65;i++){
const angle=Math.random()*Math.PI*2;
const speed=1.5+Math.random()*5;
particles.push({x,y,vx:Math.cos(angle)*speed,vy:Math.sin(angle)*speed,life:1,color:colors[Math.floor(Math.random()*colors.length)],size:1+Math.random()*2.5});
}
}
function drawFireworks(){
ctx.clearRect(0,0,innerWidth,innerHeight);
particles=particles.filter(p=>p.life>0.03);
particles.forEach(p=>{
p.x+=p.vx;p.y+=p.vy;p.vy+=0.035;p.life-=0.014;
ctx.globalAlpha=p.life;ctx.fillStyle=p.color;
ctx.beginPath();ctx.arc(p.x,p.y,p.size,0,Math.PI*2);ctx.fill();
});
ctx.globalAlpha=1;
requestAnimationFrame(drawFireworks);
}
drawFireworks();

/* CLASSIC HAPPY BIRTHDAY MUSIC */
let audioContext=null;
let musicOn=false;
let musicTimer=null;
let notePosition=0;
let activeOscillators=[];

const beat=0.38;
const melody=[
[261.63,1],[261.63,1],[293.66,2],[261.63,2],[349.23,2],[329.63,4],
[261.63,1],[261.63,1],[293.66,2],[261.63,2],[392.00,2],[349.23,4],
[261.63,1],[261.63,1],[523.25,2],[440.00,2],[349.23,2],[329.63,2],[293.66,4],
[466.16,1],[466.16,1],[440.00,2],[349.23,2],[392.00,2],[349.23,4]
];

function playBirthdayNote(){
if(!musicOn||!audioContext)return;

const [frequency,duration]=melody[notePosition%melody.length];
notePosition++;

const now=audioContext.currentTime;
const osc=audioContext.createOscillator();
const gain=audioContext.createGain();

osc.type='triangle';
osc.frequency.setValueAtTime(frequency,now);

gain.gain.setValueAtTime(0.001,now);
gain.gain.exponentialRampToValueAtTime(0.12,now+0.04);
gain.gain.exponentialRampToValueAtTime(0.001,now+duration*beat);

osc.connect(gain);
gain.connect(audioContext.destination);
osc.start(now);
osc.stop(now+duration*beat+0.05);

activeOscillators.push(osc);
osc.onended=()=>{
activeOscillators=activeOscillators.filter(item=>item!==osc);
};

musicTimer=setTimeout(playBirthdayNote,duration*beat*1000);
}

async function toggleMusic(){
const btn=document.getElementById('musicButton');

if(!musicOn){
const AudioAPI=window.AudioContext||window.webkitAudioContext;
if(!AudioAPI){
alert('Audio is not supported in this browser.');
return;
}

audioContext=audioContext||new AudioAPI();
await audioContext.resume();

musicOn=true;
notePosition=0;
btn.textContent='♫';
btn.style.background='#c64d7c';
playBirthdayNote();

}else{
musicOn=false;
clearTimeout(musicTimer);

activeOscillators.forEach(osc=>{
try{osc.stop();}catch(e){}
});
activeOscillators=[];

btn.textContent='♪';
btn.style.background='#d85c8d';
}
}
